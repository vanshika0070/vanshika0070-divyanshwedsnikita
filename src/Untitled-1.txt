/* ========================================
   REVEAL ANIMATION
======================================== */

function useReveal() {
  const ref = useRef(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          el.classList.add("in");
          observer.disconnect();
        }
      },
      { threshold: 0.15 }
    );

    observer.observe(el);

    return () => observer.disconnect();
  }, []);

  return ref;
}

function Reveal({
  as: Tag = "div",
  className = "",
  delay = 0,
  children,
  ...props
}) {
  const ref = useReveal();

  return (
    <Tag
      ref={ref}
      className={`rv ${className}`}
      style={{ transitionDelay: `${delay}ms` }}
      {...props}
    >
      {children}
    </Tag>
  );
}

function Dust() {
  const canvasRef = useRef(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const context = canvas.getContext("2d");

    const reducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    ).matches;

    let width = 0;
    let height = 0;
    let animationFrame;

    const resize = () => {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    };

    resize();

    window.addEventListener("resize", resize);

    const particles = Array.from({ length: 60 }, () => ({
      x: Math.random() * window.innerWidth,
      y: Math.random() * window.innerHeight,
      radius: Math.random() * 1.7 + 0.3,
      velocity: Math.random() * 0.25 + 0.05,
      alpha: Math.random() * 0.5 + 0.2,
      phase: Math.random() * 6,
    }));

    const draw = () => {
      context.clearRect(0, 0, width, height);

      particles.forEach((particle) => {
        if (!reducedMotion) {
          particle.y -= particle.velocity;
          particle.phase += 0.015;
          particle.x += Math.sin(particle.phase) * 0.25;

          if (particle.y < -5) {
            particle.y = height + 5;
            particle.x = Math.random() * width;
          }
        }

        context.beginPath();

        context.arc(
          particle.x,
          particle.y,
          particle.radius,
          0,
          Math.PI * 2
        );

        context.fillStyle = `rgba(232,213,168,${
          particle.alpha *
          (0.6 + 0.4 * Math.sin(particle.phase * 2))
        })`;

        context.fill();
      });

      animationFrame = window.requestAnimationFrame(draw);
    };

    draw();

    return () => {
      window.cancelAnimationFrame(animationFrame);
      window.removeEventListener("resize", resize);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      className="dust"
      aria-hidden="true"
    />
  );
}

function Countdown() {
  const [now, setNow] = useState(Date.now());

  useEffect(() => {
    const timer = window.setInterval(
      () => setNow(Date.now()),
      1000
    );

    return () => window.clearInterval(timer);
  }, []);

  const remaining = Math.max(
    0,
    W.date.getTime() - now
  );

  const units = [
    ["Days", Math.floor(remaining / 864e5)],
    ["Hours", Math.floor(remaining / 36e5) % 24],
    ["Minutes", Math.floor(remaining / 6e4) % 60],
    ["Seconds", Math.floor(remaining / 1e3) % 60],
  ];

  return (
    <section className="band">
      <Reveal className="center">
        <h2>Until the “I do”</h2>

        <div className="count">
          {units.map(([label, value]) => (
            <div key={label}>
              <span>
                {String(value).padStart(2, "0")}
              </span>
              <small>{label}</small>
            </div>
          ))}
        </div>
      </Reveal>
    </section>
  );
}

   MAIN APP
======================================== */

export default function Design() {
  const [opened, setOpened] = useState(false);
  const [envelopeOpened, setEnvelopeOpened] = useState(false);
  const [sealOpening, setSealOpening] = useState(false);
  const [muted, setMuted] = useState(false);

  const audioRef = useRef(null);
  const nextAudioRef = useRef(null);

  const fadeRef = useRef(null);
  const loopRef = useRef(null);

  const switchingRef = useRef(false);
  const openingTimerRef = useRef(null);

  /* ========================================
     LOCK SCROLL UNTIL INVITATION OPENS
  ======================================== */

  useEffect(() => {
    document.body.style.overflow = opened ? "" : "hidden";

    return () => {
      document.body.style.overflow = "";
    };
  }, [opened]);

  /* ========================================
     BACKGROUND MUSIC - SMOOTH LOOP
  ======================================== */

  const playMusic = async () => {
    if (!W.music || audioRef.current) return;

    const first = new Audio(W.music);
    const second = new Audio(W.music);

    first.preload = "auto";
    second.preload = "auto";

    first.volume = 0;
    second.volume = 0;

    audioRef.current = first;
    nextAudioRef.current = second;

    const TARGET_VOLUME = 0.4;
    const CROSSFADE_TIME = 2.5;
    const START_BEFORE_END = 4;

    try {
      await first.play();

      /* INITIAL FADE IN */

      const fadeInStart = performance.now();

      const fadeIn = () => {
        if (
          !audioRef.current ||
          audioRef.current !== first
        ) {
          return;
        }

        const progress = Math.min(
          (performance.now() - fadeInStart) / 2000,
          1
        );

        first.volume = TARGET_VOLUME * progress;

        if (progress < 1) {
          fadeRef.current = setTimeout(fadeIn, 25);
        } else {
          fadeRef.current = null;
        }
      };

      fadeIn();

      /* CROSSFADE LOOP */

      loopRef.current = setInterval(async () => {
        const current = audioRef.current;
        const next = nextAudioRef.current;

        if (
          !current ||
          !next ||
          switchingRef.current ||
          !Number.isFinite(current.duration) ||
          current.duration <= 0
        ) {
          return;
        }

        const remaining =
          current.duration - current.currentTime;

        if (remaining > START_BEFORE_END) return;

        switchingRef.current = true;

        try {
          next.currentTime = 0;
          next.volume = 0;

          await next.play();

          const startTime = performance.now();
          const duration = CROSSFADE_TIME * 1000;
          const startingVolume = current.volume;

          clearTimeout(fadeRef.current);

          const crossfade = () => {
            const progress = Math.min(
              (performance.now() - startTime) / duration,
              1
            );

            current.volume = Math.max(
              0,
              startingVolume * (1 - progress)
            );

            next.volume = Math.min(
              TARGET_VOLUME,
              TARGET_VOLUME * progress
            );

            if (progress < 1) {
              fadeRef.current = setTimeout(
                crossfade,
                25
              );
            } else {
              current.pause();
              current.currentTime = 0;
              current.volume = 0;

              audioRef.current = next;
              nextAudioRef.current = current;

              switchingRef.current = false;
              fadeRef.current = null;
            }
          };

          crossfade();

        } catch (error) {
          console.log(
            "Crossfade playback failed:",
            error
          );

          switchingRef.current = false;
        }
      }, 50);

    } catch (error) {
      console.log("Audio playback failed:", error);

      audioRef.current = null;
      nextAudioRef.current = null;

      first.pause();
      second.pause();
    }
  };

  /* ========================================
     OPEN INVITATION
  ======================================== */

  const openInvitation = () => {
    if (envelopeOpened || sealOpening) return;

    // 1. Start the seal zoom-out animation.
    setSealOpening(true);

    // 2. Open the gates after the seal disappears.
    openingTimerRef.current = setTimeout(() => {
      setEnvelopeOpened(true);

      // 3. Reveal the invitation after the gates open.
      openingTimerRef.current = setTimeout(() => {
        setOpened(true);
        playMusic();
      }, 1300);
    }, 1200);
  };

  /* ========================================
     MUSIC ON / OFF
  ======================================== */

  const toggleMusic = () => {
    const newMuted = !muted;

    if (audioRef.current) {
      audioRef.current.muted = newMuted;
    }

    if (nextAudioRef.current) {
      nextAudioRef.current.muted = newMuted;
    }

    setMuted(newMuted);
  };

  /* ========================================
     CLEANUP
  ======================================== */

  useEffect(() => {
    return () => {
      clearTimeout(openingTimerRef.current);
      clearTimeout(fadeRef.current);
      clearInterval(loopRef.current);

      audioRef.current?.pause();
      nextAudioRef.current?.pause();

      audioRef.current = null;
      nextAudioRef.current = null;
    };
  }, []);

  /* ========================================
     RETURN
  ======================================== */

  return (
    <>
      <Dust />

      <div className="grain" aria-hidden="true" />

      {/* ========================================
          IMAGE GATE SCREEN
      ======================================== */}

      <div className={`gate ${opened ? "gone" : ""}`}>
        <div className="envelope-scene">
          <div
            className={`envelope ${
              envelopeOpened ? "envelope-open" : ""
            }`}
          >
            {/* Background behind the doors */}

            <div className="envelope-back" />

            {/* TWO IMAGE DOORS */}

            <div className="envelope-pocket">
              {/* LEFT HALF */}

              <div
                className="gate-left"
                style={{
                  backgroundImage: `url("${W.gateImage}")`,
                }}
              />

              {/* RIGHT HALF */}

              <div
                className="gate-right"
                style={{
                  backgroundImage: `url("${W.gateImage}")`,
                }}
              />
            </div>

            {/* WAX SEAL */}

            <button
              className={`wax-seal ${
                sealOpening ? "seal-opening" : ""
              }`}
              onClick={openInvitation}
              disabled={sealOpening || envelopeOpened}
              aria-label="Open wedding invitation"
            >
              <span>
                {W.groom[0]}&{W.bride[0]}
              </span>
            </button>
          </div>
        </div>
      </div>

      {/* ========================================
          MUSIC CONTROL
      ======================================== */}

      {opened && W.music && (
        <button
          className="mute"
          onClick={toggleMusic}
          aria-label={
            muted ? "Turn music on" : "Turn music off"
          }
        >
          {muted ? "Sound off" : "Sound on"}
        </button>
      )}

      {/* ========================================
          MAIN WEDDING WEBSITE
      ======================================== */}

      <main>
        <Hero play={opened} />

        <Countdown />

        <OtherCelebrations />

        <Wedding />

        <Reception />
      </main>

      {/* ========================================
          FOOTER
      ======================================== */}

      <footer className="center">
        <p className="footer-note">
          "With our families' blessings, join us in celebrating a beautiful new beginning."
        </p>

        <p className="fn">
          {W.groom} <i>&</i> {W.bride}
        </p>
      </footer>
    </>
  );
}
